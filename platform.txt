# Arduino BL platform.
# Specification: https://arduino.github.io/arduino-cli/latest/platform-specification/
#
# Compile/link flags and libraries come from the board-selected precompiled package (boards.txt build.bl_libs).
# Ordinary platform builds never invoke SDK builds. Pass package flags/ files as GCC response files.
# Use -iprefix with package-relative include paths so installation roots remain relocatable.

name=Bouffalo Lab BL boards
version=0.1.0

# Toolchain: GCC RISC-V 14.3.0 (bouffalolab/toolchain, picolibc).
compiler.path={runtime.tools.gcc-riscv.path}/bin/
compiler.prefix=riscv64-unknown-elf-
compiler.c.cmd={compiler.prefix}gcc
compiler.cpp.cmd={compiler.prefix}g++
compiler.S.cmd={compiler.prefix}gcc
compiler.ar.cmd={compiler.prefix}ar
compiler.c.elf.cmd={compiler.prefix}g++
compiler.objcopy.cmd={compiler.prefix}objcopy
compiler.size.cmd={compiler.prefix}size

compiler.warning_flags=-w
compiler.warning_flags.none=-w
compiler.warning_flags.default=
compiler.warning_flags.more=-Wall
compiler.warning_flags.all=-Wall -Wextra

# Package configuration defines, include paths and forced autoconfig.h.
compiler.bl.sdk="@{build.bl_libs}/flags/defines" -iprefix "{build.bl_libs}/include/" "@{build.bl_libs}/flags/includes" -include "{build.bl_libs}/include/build/generated/autoconfig.h"
# Board USB identity, included through build.extra_flags for USB boards, following the official AVR platform.
build.usb_flags=-DUSB_VID={build.vid} -DUSB_PID={build.pid} '-DUSB_MANUFACTURER={build.usb_manufacturer}' '-DUSB_PRODUCT={build.usb_product}'
compiler.bl.arduino=-DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} {build.extra_flags}

# Package c_flags contain -Os -g; later optimization flags override them. IDE debug optimization or --optimize-for-debug selects debug.
compiler.optimization_flags=-Os
compiler.optimization_flags.release=-Os
compiler.optimization_flags.debug=-Og -g3

compiler.c.flags=-c "@{build.bl_libs}/flags/c_flags" {compiler.optimization_flags} -std=gnu99 -MMD {compiler.warning_flags}
compiler.cpp.flags=-c "@{build.bl_libs}/flags/c_flags" {compiler.optimization_flags} -std=gnu++17 -fno-rtti -fno-exceptions -MMD {compiler.warning_flags}
compiler.S.flags=-c "@{build.bl_libs}/flags/S_flags" -x assembler-with-cpp -MMD
compiler.ar.flags=rcs
compiler.c.extra_flags=
compiler.cpp.extra_flags=
compiler.S.extra_flags=
compiler.ar.extra_flags=
compiler.c.elf.extra_flags=
# Populated by library.properties ldflags, such as WiFi's flags/ld_libs_wifi.
compiler.libraries.ldflags=

recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {compiler.c.extra_flags} {compiler.bl.arduino} {includes} {compiler.bl.sdk} "{source_file}" -o "{object_file}"
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {compiler.cpp.extra_flags} {compiler.bl.arduino} {includes} {compiler.bl.sdk} "{source_file}" -o "{object_file}"
recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} {compiler.S.extra_flags} {compiler.bl.arduino} {includes} {compiler.bl.sdk} "{source_file}" -o "{object_file}"
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} {compiler.ar.extra_flags} "{archive_file_path}" "{object_file}"

# Do not whole-archive every library: ld_libs already retains the required-only-as-a-whole members via libforce_base.a.
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" "@{build.bl_libs}/flags/ld_flags" {compiler.c.elf.extra_flags} -T "{build.path}/layout.ld" "-L{build.bl_libs}/ld" "-L{build.bl_libs}/lib" "-Wl,-Map={build.path}/{build.project_name}.map" -o "{build.path}/{build.project_name}.elf" {object_files} -Wl,--start-group "{archive_file_path}" {compiler.libraries.ldflags} "@{build.bl_libs}/flags/ld_libs" -Wl,--end-group

recipe.objcopy.bin.pattern="{compiler.path}{compiler.objcopy.cmd}" -O binary "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.bin"

# bflb_fw_post_proc updates the binary boot header (length, CRC and DTS parameters) in place.
# It also generates partition.bin beside the image for upload. Tools currently support Linux only.
# Do not pass --boot2file: it copies package boot2 with read-only permissions, then modifies the copy and fails.
# Processed boot2 is byte-identical to the packaged file; upload the package copy directly.
recipe.hooks.objcopy.postobjcopy.1.pattern="{runtime.tools.bl-flash-tools.path}/bl-layout" finalize --build "{build.path}" --project "{build.project_name}" --libs "{build.bl_libs}" --tools "{runtime.tools.bl-flash-tools.path}" --dts "{build.bl_dts}"

recipe.output.tmp_file={build.project_name}.bin
recipe.output.save_file={build.project_name}.{build.variant}.bin

# Flash counts loaded sections, including initial values copied to RAM. RAM counts sections in the ram region.
recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -A "{build.path}/{build.project_name}.elf"
recipe.size.regex=^(?:\.fw_header|\.init|\.rftlv\.tool|\.ver|\.init_code|rom_text|rom_rodata|\.last_section|tcm_text|tcm_rodata|tcm_data|ram_data|ram_nocache_data)\s+([0-9]+).*
recipe.size.regex.data=^(?:tcm_text|tcm_rodata|tcm_data|ram_data|ram_bss|ram_noinit|ram_nocache_data|ram_nocache_noinit)\s+([0-9]+).*

# Serial discovery and monitor: uploads and Serial both use serial ports.
pluggable_discovery.required.0=builtin:serial-discovery
pluggable_monitor.required.serial=builtin:serial-monitor

# BLFlashCommand writes boot2, partition table and firmware at their selected addresses.
# ROM download exposes USB CDC; retain --interface=uart, although baud rate has no physical effect on USB.
# --warm_reset starts the new application. A 1200 bps touch request remains in the HBN domain.
# A normal --reset would reenter ROM; Core startup clears the retained request.
# BLFlashCommand modifies its directory. bl-layout runs a build-local flash-runner copy, preserving read-only installations.
tools.bl_flash.path={runtime.tools.bl-flash-tools.path}/bouffalo_flash_cube
tools.bl_flash.cmd=BLFlashCommand-ubuntu
tools.bl_flash.upload.params.verbose=
tools.bl_flash.upload.params.quiet=
tools.bl_flash.upload.pattern="{runtime.tools.bl-flash-tools.path}/bl-layout" upload --build "{build.path}" --project "{build.project_name}" --libs "{build.bl_libs}" --tools "{runtime.tools.bl-flash-tools.path}" --port "{upload.port.address}" --speed "{upload.speed}"

# One selected layout snapshot drives linking, post-processing, sizing and upload.
build.bl_partition_file=
recipe.hooks.prebuild.1.pattern="{runtime.tools.bl-flash-tools.path}/bl-layout" prepare --build "{build.path}" --sketch "{build.source.path}" --libs "{build.bl_libs}" --default "{build.bl_partition}" --boot2 "{build.bl_boot2}" --scheme "{build.bl_partition_scheme}" "--file={build.bl_partition_file}"
recipe.advanced_size.pattern="{runtime.tools.bl-flash-tools.path}/bl-layout" size --build "{build.path}" --project "{build.project_name}" --size-tool "{compiler.path}{compiler.size.cmd}" --ram {upload.maximum_data_size}
recipe.hooks.savehex.postsavehex.1.pattern="{runtime.tools.bl-flash-tools.path}/bl-layout" export --build "{build.path}" --project "{build.project_name}" --destination "{sketch_path}/bl-flash"
