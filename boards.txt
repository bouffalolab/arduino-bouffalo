# Boards for the Arduino BL platform.
# Specification: https://arduino.github.io/arduino-cli/latest/platform-specification/#boardstxt

menu.debuglog=Debug log
menu.PartitionScheme=Partition Scheme

# UNO R4 WiFi with a BL616CL coprocessor; sketches selected for this board run on BL616CL.
unor4wifi_bl616cl.name=UNO R4 WiFi (BL616CL coprocessor)
unor4wifi_bl616cl.build.core=bouffalo
unor4wifi_bl616cl.build.variant=unor4wifi_bl616cl
unor4wifi_bl616cl.build.board=UNOR4WIFI_BL616CL
unor4wifi_bl616cl.build.mcu=bl616cl
# CPU clock: the dependency package selects the 320 MHz WIFIPLL in system_clock_init.
unor4wifi_bl616cl.build.f_cpu=320000000L
unor4wifi_bl616cl.build.extra_flags={build.usb_flags} {build.debug_log_flags}
# USB CDC identity reported by Serial.
unor4wifi_bl616cl.build.vid=0x349B
unor4wifi_bl616cl.build.pid=0x0011
unor4wifi_bl616cl.build.usb_manufacturer="Bouffalolab"
unor4wifi_bl616cl.build.usb_product="BL616CL Arduino UNO WiFi R4 CMSIS-DAP"
# Identify this board during port discovery.
unor4wifi_bl616cl.upload_port.0.vid=0x349B
unor4wifi_bl616cl.upload_port.0.pid=0x0011
# Precompiled dependency package for the selected chip.
unor4wifi_bl616cl.build.bl_libs={runtime.tools.bl616cl-libs.path}
# Image resources come from the package boot/ directory. The DTS uses SDK development-board factory parameters; crystal and RF parameters for the replacement module remain unverified.
unor4wifi_bl616cl.build.bl_boot2=boot2.bin
unor4wifi_bl616cl.build.bl_partition=partition_cfg_4M_lfs_v1.toml
unor4wifi_bl616cl.build.bl_dts=bl_factory_params_IoTKitA_auto.dts
# Two adjacent 1.5 MiB firmware slots, then 416 KiB littlefs. The linker ram region contains 223 KiB.
unor4wifi_bl616cl.upload.maximum_size=1572864
unor4wifi_bl616cl.upload.maximum_data_size=228352
# Upload over the onboard USB port: select the running Serial port; 1200 bps touch resets into ROM download mode.
# The ROM exposes its download port at the same USB location. Select that port directly when already in ROM mode.
unor4wifi_bl616cl.upload.tool.serial=bl_flash
unor4wifi_bl616cl.upload.protocol=serial
unor4wifi_bl616cl.upload.speed=2000000
unor4wifi_bl616cl.upload.use_1200bps_touch=true
unor4wifi_bl616cl.upload.wait_for_upload_port=true
# Addresses from partition_cfg_4M_lfs_v1.toml: partition table 0xE000, firmware 0x10000; boot2 is fixed at 0x0.
unor4wifi_bl616cl.upload.bl_partition_address=0xE000
unor4wifi_bl616cl.upload.bl_fw_address=0x10000
# Debug logging defaults to off, discarding printf and SDK output. When enabled, use uart2 and the variant PIN_DEBUG_LOG_TX/RX pins at 2000000 baud. Keep release logging off.
unor4wifi_bl616cl.menu.debuglog.off=Off
unor4wifi_bl616cl.menu.debuglog.off.build.debug_log_flags=
unor4wifi_bl616cl.menu.debuglog.uart2=uart2 (J6 IO41 TX, IO42 RX, 2000000 baud)
unor4wifi_bl616cl.menu.debuglog.uart2.build.debug_log_flags=-DARDUINO_BL_DEBUG_LOG_BAUD=2000000

# Custom uses partitions.toml in the sketch directory; an explicit CLI path takes priority.
unor4wifi_bl616cl.menu.PartitionScheme.default=Default (dual 1.5 MiB, FS 416 KiB)
unor4wifi_bl616cl.menu.PartitionScheme.default.build.bl_partition_scheme=default
unor4wifi_bl616cl.menu.PartitionScheme.custom=Custom (sketch partitions.toml)
unor4wifi_bl616cl.menu.PartitionScheme.custom.build.bl_partition_scheme=custom
